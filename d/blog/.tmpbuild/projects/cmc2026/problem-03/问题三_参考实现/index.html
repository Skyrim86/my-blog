机器狗客户端 -- 问题三（全向源）：有界不确定性下的主动定位 + 全局联合路径规划

算法要点（本文件即论文问题三的正式算法实现，附录 B 为其逐字副本）
 1. **覆盖构型**：中心 + 6 环点 @ ρ = 1300 m（部署取值），解析覆盖半径
    c(6,1300) = 936.4834 m <= R_MIN = 1000 m（余量 63.52 m）=> 任意位置的源必被至少
    一个采样点听到，"全部采样点 no_signal => 该频道无源"严格成立。构型族的一般式
    c(k,ρ) = max{ ρ/(2cos(π/k)), sqrt(1800^2 + ρ^2 - 3600ρcos(π/k)) }，可行窗口是双侧的，
    时间最优在窗口下端（论文 Sec 5.1(3)）；`Q3V_RING`/`Q3V_K` 可切换构型做对照。
 2. **信念（有界不确定性）**：每次测向给出 +/-1° 楔形约束，一次 direction 还意味着距离
    <= R_MAX = 1500 m。可行域 F = 场地圆盘 ∩ 圆盘(S_i, 1500) ∩ 楔形(S_i, svd_i, +/-1°)
    用凸多边形半平面精确裁剪（略大于真实可行域，保守），取最小包围圆（Welzl）得
    Chebyshev 圆心 c 与半径 R*（不确定度度量）。
 3. **观测点选择（谨慎贪心）**：候选点 p 的得分 = max_{q∈F 代表点} R*(F ∩ 楔形(p,
    bearing(p,q))) + λ*|pos->p->c|，即对"真值位于 F 内何处"取最坏情形，再计入实际移动
    代价；候选点须满足 max_{q∈F}|p-q| <= 950 m < R_MIN（该处必有信号，信念保持凸交集）。

 4. **路线**：固定覆盖骨架（自出发点对覆盖点**一次性**求开链 TSP）+ 源按**最小插入代价**
    并入剩余路线，每步对整条剩余路线做 2-opt/Or-opt 重排；R* <= 120 m 后不再在覆盖点
    重复观测该频道。无信息测量直接跳过（可行域最近点 >1500 m，或最乐观后验收缩不足
    30 m）。
 5. **清除链**：R* <= 16 m 时在区域中心 /clear -- 真值必在区域内、与 c 距离 <= R* < 20 m
    （清除半径），**必然命中**；否则依次走"中心复测 -> 垂直视线偏移 +/-100 m 复测 ->
    可行域格点清除 -> 单线补线（沿射线前进 a + 侧偏 w，几何上必在接收半径内）->
    沿视线用"有信号->无信号"翻转点二分 -> 追踪式逼近 -> 局部 14 m 网格兜底"。
 6. **工程补强**：测量/清除去重缓存（题目明示"同一地点重复检测误差不变"，同点同频道结果
    确定，重复动作直接复用）、无信息测量跳过、判无源前"必须测遍全部覆盖点"的完备性护栏。

所有开关都可由环境变量 `Q3V_*` 关闭以做消融（论文 Sec 7.5 的消融档即由它们产生）。

方法参考三篇主动定位文献：

# ==================== 基础几何 ====================
def ring_points(radius, n, phase=0.0):
    return [(radius * math.cos(phase + 2 * math.pi * i / n),
             radius * math.sin(phase + 2 * math.pi * i / n)) for i in range(n)]


def dist(a, b):
    return math.hypot(a[0] - b[0], a[1] - b[1])


def angdiff(a, b):
    d = abs(a - b) % 360.0
    return min(d, 360.0 - d)


def uvec(deg):
    r = math.radians(deg)
    return (math.cos(r), math.sin(r))


def perp(deg):
    r = math.radians(deg)
    return (-math.sin(r), math.cos(r))


def bearing(p, q):
    return math.degrees(math.atan2(q[1] - p[1], q[0] - p[0]))


def clamp(p, r=ARENA_R + 120.0):
    d = math.hypot(p[0], p[1])
    return (p[0], p[1]) if d <= r else (p[0] * r / d, p[1] * r / d)


def circ180(a, b):
    """两条视线方向的无向夹角（0..90）。"""
    d = abs(a - b) % 180.0
    return min(d, 180.0 - d)


def best_pair_angle(rays):
    """已获视线对的最大交会角（°）；单线/空返回 0。"""
    best = 0.0
    for i in range(len(rays)):
        for j in range(i + 1, len(rays)):
            best = max(best, circ180(rays[i][2], rays[j][2]))
    return best


COVER_POINTS = [(0.0, 0.0)] + ring_points(COVER_R, COVER_N, 0.0)


def coverage_radius(points=COVER_POINTS, R=ARENA_R, step=30.0):
    worst = 0.0
    k = int(R / step) + 1
    for i in range(-k, k + 1):
        for j in range(-k, k + 1):
            x, y = i * step, j * step
            if x * x + y * y > R * R:
                continue
            d = min(math.hypot(x - p[0], y - p[1]) for p in points)
            if d > worst:
                worst = d
    return worst


# ==================== 凸区域：半平面/圆裁剪 + 最小包围圆 ====================
def _wedge_halfplanes(S, theta, half=DELTA):
    """+/-half° 楔形的两条半平面（法向指向楔形内部）。"""
    a1 = math.radians(theta - half)
    a2 = math.radians(theta + half)
    return [(S, (-math.sin(a1), math.cos(a1))), (S, (math.sin(a2), -math.cos(a2)))]


def _clip_hp(poly, q, n):
    """Sutherland-Hodgman：保留 n*(p-q) >= 0 的一侧。"""
    if not poly:
        return poly
    out = []
    m = len(poly)
    for i in range(m):
        a, b = poly[i], poly[(i + 1) % m]
        da = n[0] * (a[0] - q[0]) + n[1] * (a[1] - q[1])
        db = n[0] * (b[0] - q[0]) + n[1] * (b[1] - q[1])
        if da >= -1e-9:
            out.append(a)
        if (da > 1e-9 > db) or (db > 1e-9 > da):
            t = da / (da - db)
            out.append((a[0] + t * (b[0] - a[0]), a[1] + t * (b[1] - a[1])))
    return out


def _clip_circle(poly, c, r):
    """保留 |p-c| <= r 的凸多边形部分（精确圆弧裁剪，处理穿越弦）。"""
    if not poly:
        return poly
    out = []
    m = len(poly)
    cx, cy = c
    r2 = r * r
    for i in range(m):
        a, b = poly[i], poly[(i + 1) % m]
        da = math.hypot(a[0] - cx, a[1] - cy) - r
        if da <= 1e-12:
            out.append(a)
        dx, dy = b[0] - a[0], b[1] - a[1]
        A = dx * dx + dy * dy
        if A < 1e-18:
            continue
        fx, fy = a[0] - cx, a[1] - cy
        B = 2.0 * (fx * dx + fy * dy)
        C = fx * fx + fy * fy - r2
        disc = B * B - 4.0 * A * C
        if disc <= 0.0:
            continue
        sq = math.sqrt(disc)
        for t in sorted(((-B - sq) / (2.0 * A), (-B + sq) / (2.0 * A))):
            if 1e-12 < t < 1.0 - 1e-12:
                out.append((a[0] + t * dx, a[1] + t * dy))
    return out


DISK_N = 120


def _disk_halfplanes(c, r):
    """用外接正 DISK_N 边形（含圆，保守）近似圆盘约束，化为半平面。"""
    R = r / math.cos(math.pi / DISK_N)
    out = []
    for k in range(DISK_N):
        ph = 2.0 * math.pi * k / DISK_N
        nx, ny = math.cos(ph), math.sin(ph)
        out.append(((c[0] + R * nx, c[1] + R * ny), (-nx, -ny)))
    return out


def _circum(a, b, c):
    ax, ay = a
    bx, by = b
    cx, cy = c
    d = 2.0 * (ax * (by - cy) + bx * (cy - ay) + cx * (ay - by))
    if abs(d) < 1e-12:
        best = (a[0], a[1], 0.0)
        for p, q in ((a, b), (b, c), (a, c)):
            r = dist(p, q) / 2.0
            if r > best[2]:
                best = ((p[0] + q[0]) / 2.0, (p[1] + q[1]) / 2.0, r)
        return best
    ux = ((ax * ax + ay * ay) * (by - cy) + (bx * bx + by * by) * (cy - ay)
          + (cx * cx + cy * cy) * (ay - by)) / d
    uy = ((ax * ax + ay * ay) * (cx - bx) + (bx * bx + by * by) * (ax - cx)
          + (cx * cx + cy * cy) * (bx - ax)) / d
    return (ux, uy, math.hypot(ax - ux, ay - uy))


def mec(points, seed=20260912):
    """Welzl 最小包围圆：返回 (cx, cy, R)。"""
    pts = list(points)
    if not pts:
        return None
    if len(pts) == 1:
        return (pts[0][0], pts[0][1], 0.0)
    random.Random(seed).shuffle(pts)

    def inside(p, c):
        return math.hypot(p[0] - c[0], p[1] - c[1]) <= c[2] + 1e-7

    c = None
    for i, p in enumerate(pts):
        if c is not None and inside(p, c):
            continue
        c = (p[0], p[1], 0.0)
        for j in range(i):
            q = pts[j]
            if inside(q, c):
                continue
            c = ((p[0] + q[0]) / 2.0, (p[1] + q[1]) / 2.0, dist(p, q) / 2.0)
            for k in range(j):
                r = pts[k]
                if inside(r, c):
                    continue
                c = _circum(p, q, r)
    return c


# ==================== 有界不确定性信念 ====================
def region(rays):
    """由测向约束精确构造可行域。返回 dict(poly, cons, c, R) 或 None。

    F = 场地圆盘 ∩_i 圆盘(S_i, R_MAX) ∩_i 楔形(S_i, svd_i, +/-1°)。
    圆盘用外接正多边形（含圆）化为半平面，与楔形一起做凸多边形半平面裁剪；
    区域略大于真实可行域（保守），故 R*<=TIGHT 时清除必然命中。
    """
    if not rays:
        return None
    cons = list(_disk_halfplanes((0.0, 0.0), ARENA_R))
    for (x, y, th) in rays:
        cons.extend(_wedge_halfplanes((x, y), th))
        cons.extend(_disk_halfplanes((x, y), R_MAX + 0.5))
    poly = [(-BOX, -BOX), (BOX, -BOX), (BOX, BOX), (-BOX, BOX)]
    for (q, n) in cons:
        poly = _clip_hp(poly, q, n)
        if len(poly) < 3:
            return None
    c = mec(poly)
    if c is None:
        return None
    return {"poly": poly, "cons": cons, "c": (c[0], c[1]), "R": c[2]}


def _inside(reg, p):
    for (q, n) in reg["cons"]:
        if n[0] * (p[0] - q[0]) + n[1] * (p[1] - q[1]) < -1e-7:
            return False
    if p[0] * p[0] + p[1] * p[1] > ARENA_R * ARENA_R + 1e-6:
        return False
    return True


def _region_after(reg, S, theta):
    """当前可行域 ∩ 楔形(S, theta+/-1°) 的 (c, R)。"""
    poly = reg["poly"]
    for (q, n) in _wedge_halfplanes(S, theta):
        poly = _clip_hp(poly, q, n)
        if len(poly) < 3:
            return None
    cc = mec(poly)
    if cc is None:
        return None
    return ((cc[0], cc[1]), cc[2])


def _spread(reg, k=10):
    """从可行域取 k 个代表点（顶点 + 边中点），用于最坏情形评估。"""
    poly = reg["poly"]
    pts = list(poly)
    m = len(poly)
    for i in range(m):
        a, b = poly[i], poly[(i + 1) % m]
        pts.append(((a[0] + b[0]) / 2.0, (a[1] + b[1]) / 2.0))
    if len(pts) <= k:
        return pts
    st = len(pts) / float(k)
    return [pts[int(i * st)] for i in range(k)]


def _poly_min_dist(p, poly):
    """点到凸多边形的距离（内部为 0）。"""
    inside = True
    m = len(poly)
    # 简单内外判定：对每条边取叉积符号一致性
    sgn = 0
    for i in range(m):
        a, b = poly[i], poly[(i + 1) % m]
        cr = (b[0] - a[0]) * (p[1] - a[1]) - (b[1] - a[1]) * (p[0] - a[0])
        if cr > 1e-9:
            if sgn < 0:
                inside = False
                break
            sgn = 1
        elif cr < -1e-9:
            if sgn > 0:
                inside = False
                break
            sgn = -1
    if inside:
        return 0.0
    best = float("inf")
    for i in range(m):
        a, b = poly[i], poly[(i + 1) % m]
        dx, dy = b[0] - a[0], b[1] - a[1]
        L2 = dx * dx + dy * dy
        if L2 < 1e-18:
            d = dist(p, a)
        else:
            t = max(0.0, min(1.0, ((p[0] - a[0]) * dx + (p[1] - a[1]) * dy) / L2))
            d = math.hypot(p[0] - a[0] - t * dx, p[1] - a[1] - t * dy)
        if d < best:
            best = d
    return best


def predicted_R(reg, p):
    """在 p 处若获得一条朝向可行域中心的测向线，预测后验可行域半径 R*。"""
    cx, cy = reg["c"]
    if math.hypot(p[0] - cx, p[1] - cy) < 1e-9:
        return None
    nr = _region_after(reg, p, bearing(p, (cx, cy)))
    return None if nr is None else nr[1]


def choose_view(reg, pos, avoid=None):
    """谨慎贪心选点：最小化「最坏情形后验 R*」+ λ*移动距离。"""
    cx, cy = reg["c"]
    R = reg["R"]
    samples = _spread(reg, 10)
    reach = max(0.0, SAFE_RANGE ** 2 - R * R) ** 0.5
    cands = []
    for L in (max(45.0, 1.6 * R), R + 200.0, 0.55 * reach, 0.85 * reach):
        L = min(L, reach)
        if L < 45.0:
            continue
        for k in range(12):
            ph = 2.0 * math.pi * k / 12.0
            p = (cx + L * math.cos(ph), cy + L * math.sin(ph))
            if p[0] * p[0] + p[1] * p[1] <= ARENA_R * ARENA_R:
                cands.append(p)
    dp = dist(pos, (cx, cy))
    if dp > 1.0:
        ux, uy = (cx - pos[0]) / dp, (cy - pos[1]) / dp
        for L in (max(60.0, dp - R - 150.0), dp * 0.55):
            p = (cx - L * ux, cy - L * uy)
            if p[0] * p[0] + p[1] * p[1] <= ARENA_R * ARENA_R:
                cands.append(p)
    cands.append((cx, cy))
    uniq = []
    for p in cands:
        p = clamp(p)
        if p[0] * p[0] + p[1] * p[1] > ARENA_R * ARENA_R:
            continue
        if avoid is not None and dist(p, avoid) < 25.0:
            continue
        if any(dist(p, u) < 20.0 for u in uniq):
            continue
        uniq.append(p)
    best = None
    for p in uniq:
        if max(dist(p, q) for q in samples) > SAFE_RANGE:
            continue
        worst = 0.0
        for q in samples:
            nr = _region_after(reg, p, bearing(p, q))
            if nr is None:
                continue
            if nr[1] > worst:
                worst = nr[1]
        # 观测后仍需前往可行域（清源），故计入 pos->p->c 的整段移动代价
        travel = dist(pos, p) + dist(p, (cx, cy))
        score = worst + TUNE["LAMBDA"] * travel
        if best is None or score < best[0]:
            best = (score, p, worst)
    if best is None:
        return None
    return best[1]

def _path_cost(order, nodes, pos_of, start):
    if not order:
        return 0.0
    c = dist(start, pos_of(nodes[order[0]]))
    for a, b in zip(order, order[1:]):
        c += dist(pos_of(nodes[a]), pos_of(nodes[b]))
    return c


def plan_order(start, nodes, pos_of):
    """最近邻构造 + 2-opt + Or-opt，返回访问次序（nodes 的下标列表）。"""
    n = len(nodes)
    if n == 0:
        return []
    if n == 1:
        return [0]
    left = list(range(n))
    order, cur = [], start
    while left:
        k = min(left, key=lambda i: dist(pos_of(nodes[i]), cur))
        order.append(k)
        cur = pos_of(nodes[k])
        left.remove(k)
    best = _path_cost(order, nodes, pos_of, start)
    improved = True
    while improved:
        improved = False
        for i in range(n - 1):
            for j in range(i + 1, n):
                cand = order[:i] + order[i:j + 1][::-1] + order[j + 1:]
                c = _path_cost(cand, nodes, pos_of, start)
                if c < best - 1e-9:
                    order, best, improved = cand, c, True
        for seg_len in (1, 2, 3):
            for i in range(0, n - seg_len + 1):
                seg = order[i:i + seg_len]
                rest = order[:i] + order[i + seg_len:]
                for k in range(len(rest) + 1):
                    if k == i:
                        continue
                    cand = rest[:k] + seg + rest[k:]
                    c = _path_cost(cand, nodes, pos_of, start)
                    if c < best - 1e-9:
                        order, best, improved = cand, c, True
    return order

    def _post(self, path, payload):
        data = json.dumps(payload, ensure_ascii=False).encode("utf-8")
        last = None
        for attempt in range(self.max_attempts):
            if self.stop_flag is not None and self.stop_flag.is_set():
                raise KeyboardInterrupt("stopped")
            try:
                if self._conn is None:
                    host, port = self._split(self.base)
                    self._conn = http.client.HTTPConnection(host, port, timeout=10)
                self._conn.request("POST", path, body=data,
                                   headers={"Content-Type": "application/json"})
                r = self._conn.getresponse()
                raw = r.read()
                body = json.loads(raw.decode("utf-8"))
                if r.status != 200 or not body.get("accepted"):
                    last = RuntimeError("HTTP %d: %s" % (r.status, body))
                    self._close()
                else:
                    return body
            except KeyboardInterrupt:
                raise
            except Exception as e:
                last = e
                self._close()
            time.sleep(0.2 * (attempt + 1))
        raise RuntimeError("%s 连续失败: %s" % (path, last))

    def _close(self):
        try:
            if self._conn is not None:
                self._conn.close()
        except Exception:
            pass
        self._conn = None

        self._counter += 1
        return "%s-%d" % (tag, self._counter)

        mx = r.get("max_real_duration_s", 1200)
        if self.remaining_real is not None and self.remaining_real < mx - 5:
            self.deadline = self.t_start + max(0, self.remaining_real - 15)
        self.vt = r["virtual_time_s"]
        self.note("进入：remaining_real_duration_s=%s" % self.remaining_real)

    def measure(self, x, y, ch):
        x, y, ch = round(x, 3), round(y, 3), int(ch)
        if ch in self.cleared:               # 已清除频道不可能再现信号
            return "no_signal", None
        key = (ch, x, y)
        if V["DEDUP"] and key in self._meas_cache:
            # 同点同频道结果确定（题目：同一地点重复检测误差不变）-> 复用，省 6 s；
            # 该点的射线/no_signal 早在首次测量时已进信念，无需重复入账。
            return self._meas_cache[key]
        p = self._payload(self._rid("measure"))
        p["position"] = {"x": x, "y": y}
        p["channel"] = ch
        r = self._post("/measure", p)
        self.pos, self.channel, self.vt = (x, y), ch, r["virtual_time_s"]
        self._reg.pop(ch, None)
        res, svd = r.get("measure_result"), r.get("svd_deg")
        if res == "no_signal":
            self.nosig.setdefault(ch, []).append((x, y))
        elif res == "direction" and svd is not None:
            self.rays.setdefault(ch, []).append((x, y, float(svd)))
        self._meas_cache[key] = (res, svd)
        return res, svd

    def clear(self, x, y, ch):
        x, y, ch = round(x, 3), round(y, 3), int(ch)
        key = (ch, x, y)
        if V["DEDUP"] and key in self._clear_cache:
            return "no_target_in_range"     # 同点同频道已失败 -> 不再重试（失败也仅 3 s，但省一次往返）
        p = self._payload(self._rid("clear"))
        p["position"] = {"x": x, "y": y}
        p["channel"] = ch
        r = self._post("/clear", p)
        self.pos, self.vt = (x, y), r["virtual_time_s"]
        res = r.get("clear_result")
        if res == "success":
            self.cleared.add(ch)
            self.results.append((ch, x, y))
            self._reg.pop(ch, None)
        else:
            self._clear_cache.add(key)
        return res

    def exit_(self):
        try:
            r = self._post("/exit", self._payload(self._rid("exit")))
            reason = r.get("exit_reason")
        finally:
            self._close()
        self.entered = False
        return reason

    # ==================== 信念 ====================
    def _region(self, ch):
        if ch in self._reg:
            return self._reg[ch]
        reg = region(self.rays.get(ch, []))
        self._reg[ch] = reg
        return reg

    def _routable(self, ch):
        reg = self._region(ch)
        return reg is not None and reg["R"] <= TUNE["STOP_R"]

    def _pending(self):
        """尚未定论、且可行域还不够小（需继续在覆盖点观测）的频道。"""
        out = []
        for ch in range(1, MAX_CH + 1):
            if ch in self.cleared or ch in self.no_source:
                continue
            if not self._routable(ch):
                out.append(ch)
        return out

    def _score(self, ch):
        """U(q)=I/(αC+ε)：优先未观测频道，其次沿用当前频道（省换频）。"""
        I = 0.35 if self.rays.get(ch) else 1.0
        C = 0.0 if ch == self.channel else 1.0
        return I / (C + 0.05)

    # ==================== 覆盖扫描 ====================
    def _scan_point(self, p, only_unseen=False):
        if V["ADAPT"]:
            pend = self._pending()
        else:                      # 消融：不做自适应按频道判空，每个点扫全部 20 频道
            pend = [ch for ch in range(1, MAX_CH + 1) if ch not in self.cleared]
        if only_unseen:
            pend = [ch for ch in pend if not self.rays.get(ch)]
        if not pend:
            return
        pend.sort(key=self._score, reverse=True)
        self.note("扫描点(%.0f,%.0f)：%d 个待定频道%s"
                  % (p[0], p[1], len(pend), "（仅未检出）" if only_unseen else ""))
        for ch in pend:
            if not self._budget_ok():
                return
            if ch in self.cleared:
                continue
            reg = self._region(ch)
            if reg is not None:
                # 该点不可能收到信号（超出接收半径上界）-> 测了也只有 no_signal
                if _poly_min_dist(p, reg["poly"]) > TUNE["SKIP_FAR"]:
                    continue
                # 几何上几乎不缩小可行域 -> 省下这次测量：
                #   pr 是「新视线恰好指向可行域中心」时的最乐观后验 R*；
                #   连最乐观情形都缩不动 MIN_SHRINK 米，实测更不会有用。
                #   （`pr > REDUNDANT*R*` 恒不成立，等于没跳，故补这一条。）
                pr = predicted_R(reg, p)
                if pr is not None:
                    if pr > TUNE["REDUNDANT"] * reg["R"]:
                        continue
                    if V["MINSHRINK"] > 0 and pr > reg["R"] - V["MINSHRINK"]:
                        continue
            res, svd = self.measure(p[0], p[1], ch)
            # 完备性护栏用：记录该频道确实在这个覆盖点上测过
            self._cover_hits.setdefault(ch, set()).add((round(p[0], 1), round(p[1], 1)))
            if res == "near":
                self.note("频道%d 近距离信号，直接清除" % ch)
                self.clear(p[0], p[1], ch)

    # ==================== 局部主动定位与保证清除 ====================
    def _resolve(self, ch):
        """主动观测 -> 收缩可行域 -> 保证清除；失败转复测/射线推进/局部网格。"""
        for _ in range(VIEW_MAX):
            if ch in self.cleared or not self._budget_ok():
                return
            reg = self._region(ch)
            if reg is None:
                break
            if reg["R"] <= TUNE["TIGHT"]:
                break
            if reg["R"] <= TUNE["NEAR_FIX"]:
                # 已相当紧：直奔可行域中心复测即可把 R* 压到 TIGHT 以内
                p = reg["c"]
            else:
                # 谨慎贪心：只在几何确实差时另选观测点（含移动代价）
                p = choose_view(reg, self.pos, avoid=self.pos) if V["VIEW"] else None
                if p is None:
                    p = reg["c"]
            res, svd = self.measure(p[0], p[1], ch)
            if res == "near":
                self.clear(p[0], p[1], ch)
                return
            nr = self._region(ch)
            self.note("频道%d 观测(%.0f,%.0f)：R* %.0f->%.0f m"
                      % (ch, p[0], p[1], reg["R"], nr["R"] if nr else -1),
                      pos=list(p), channel=ch, marker="observe")
        if ch in self.cleared:
            return
        # 保证清除（R*<=TIGHT）：真值必在可行域内，与 Chebyshev 圆心距离 <CLEAR_R
        for _ in range(3):
            reg = self._region(ch)
            if reg is None:
                break
            if reg["R"] <= TUNE["TIGHT"]:
                self.note("频道%d 保证清除：R*=%.1f m -> (%.1f,%.1f)"
                          % (ch, reg["R"], reg["c"][0], reg["c"][1]),
                          pos=list(reg["c"]), channel=ch, marker="estimate")
                if self.clear(reg["c"][0], reg["c"][1], ch) == "success":
                    return
            # 就近复测：短距离约束能把 R* 迅速压小
            c = reg["c"]
            if not self._budget_ok():
                break
            res, svd = self.measure(c[0], c[1], ch)
            if res == "near":
                self.clear(c[0], c[1], ch)
                return
        if ch in self.cleared:
            return
        # -- 清除链第二级：在 c1 沿**垂直视线方向偏移 +/-100 m** 复测，重新交会得 (c2,R*_2) --
        # 依据（论文 Sec 5.4 定理 2 与表 5）：一级良态点对给出 (c1,R*_1)（R*_1 中位 28.1 m、
        # 最大 39.6 m，不足以直接清除）；在 c1 复测并加一次垂直偏移复测后，实测
        # R*_2 <= 2.1383 m、||c2-G|| <= 1.9029 m（N=4000）=> 在 c2 处 /clear **必然命中**。
        # 只测 c1 时 R*_2 最大 19.6674 m（余量仅 0.33 m），故垂直偏移复测不可省。
        if V["OFFSET"]:
            for _ in range(2):
                if ch in self.cleared or not self._budget_ok():
                    return
                reg = self._region(ch)
                if reg is None:
                    break
                if reg["R"] <= TUNE["TIGHT"]:
                    if self.clear(reg["c"][0], reg["c"][1], ch) == "success":
                        return
                    break
                c1 = reg["c"]
                lns = self.rays.get(ch, [])
                svd = lns[-1][2] if lns else bearing(self.pos, c1)
                nrm = perp(svd)
                offs = [V["OFFSET_D"], -V["OFFSET_D"]]
                offs.sort(key=lambda o: math.hypot(c1[0] + nrm[0] * o - self.pos[0],
                                                   c1[1] + nrm[1] * o - self.pos[1]))
                for off in offs:
                    if not self._budget_ok():
                        break
                    p = (c1[0] + nrm[0] * off, c1[1] + nrm[1] * off)
                    res2, svd2 = self.measure(p[0], p[1], ch)
                    if res2 == "near":
                        self.clear(p[0], p[1], ch)
                        return
                    st = self._region(ch)
                    if st is None:
                        continue
                    if self.clear(st["c"][0], st["c"][1], ch) == "success":
                        return
        if ch in self.cleared:
            return
        # -- 清除链第三级（可选层）：第二问最优设计的测点族（ρ,ψ 族）补第二条视线 --
        if (V["Q2FIRST"] and not self._has_good_pair(ch)
                and self._acq_n.get(ch, 0) < 2):
            self._acq_n[ch] = self._acq_n.get(ch, 0) + 1
            if self._q2_lens_ray(ch) and (ch in self.cleared or V["Q2FIRST"] == 2):
                return
        # -- 清除链第三级：单线频道沿「前进 a + 侧偏 w」补第二条视线 --
        if (V["PERP"] and V["Q2FIRST"] != 2 and not self._has_good_pair(ch)
                and self._acq_n.get(ch, 0) < 2):
            self._acq_n[ch] = self._acq_n.get(ch, 0) + 1
            if self._perp_ray(ch):
                return
        # 保证清除（R*<=TIGHT）：真值必在可行域内，与 Chebyshev 圆心距离 <CLEAR_R
        for _ in range(3):
            reg = self._region(ch)
            if reg is None:
                break
            if reg["R"] <= TUNE["TIGHT"]:
                self.note("频道%d 保证清除：R*=%.1f m -> (%.1f,%.1f)"
                          % (ch, reg["R"], reg["c"][0], reg["c"][1]),
                          pos=list(reg["c"]), channel=ch, marker="estimate")
                if self.clear(reg["c"][0], reg["c"][1], ch) == "success":
                    return
            # 就近复测：短距离约束能把 R* 迅速压小
            c = reg["c"]
            if not self._budget_ok():
                break
            res, svd = self.measure(c[0], c[1], ch)
            if res == "near":
                self.clear(c[0], c[1], ch)
                return
        if ch in self.cleared:
            return
        # -- 清除链第四级：可行域格点清除（poly 包含 真值 => 铺满必命中）--
        if V["REG"] and self._region_clear(ch):
            return
        # -- 清除链第五级：沿视线二分定出源距再清除 --
        if V["BRACKET"] and self._ray_bracket_clear(ch):
            return
        if self._home_clear(ch):
            return
        reg = self._region(ch)
        if reg is not None and ch not in self._grid_done:
            self._grid_done.add(ch)          # 网格兜底每频道只做一次，避免反复烧时间
            self.note("频道%d 转局部网格兜底" % ch)
            if self._grid_clear(ch, reg["c"][0], reg["c"][1], rmax=GRID_RMAX):
                return
        self._clear_failed.add(ch)
        self.note("频道%d 本轮未能清除" % ch)

    def _has_good_pair(self, ch, min_ang=MIN_CROSS):
        return best_pair_angle(self.rays.get(ch, [])) >= min_ang

    def _q2_lens_ray(self, ch):
        """补第二条视线：第二问的"接收保证透镜 + 最小期望直径"测点族。

        取 (rho, psi) = (1000, 38°)、(800, 38°)、(700, 34°)、(600, 26°)，从最近一个
        已测视线的**测点** S 出发、按 theta +/- psi 方向取点（第二问结论：这些点落在接收保证
        透镜 F_Q 内，故 r_1 <= 1500 m 时必在程，交会角约 50°~70°）。
        仅作对照/兜底层，默认关闭（`Q3V_Q2FIRST`）。
        """
        rays = self.rays.get(ch, [])
        if not rays:
            return False
        x, y, th = rays[-1]
        S = (x, y)
        cands = []
        for (rho, psi_deg) in ((1000.0, 38.0), (800.0, 38.0), (700.0, 34.0), (600.0, 26.0)):
            for sgn in (1.0, -1.0):
                ang = (th + sgn * psi_deg) * math.pi / 180.0
                p = clamp((S[0] + rho * math.cos(ang), S[1] + rho * math.sin(ang)))
                cands.append((dist(p, self.pos), p))
        cands.sort(key=lambda t: t[0])
        tried, seen = 0, []
        for (_d, p) in cands:
            if tried >= 4 or not self._budget_ok():
                break
            if any(dist(p, q) < 60.0 for q in seen):
                continue
            seen.append(p)
            tried += 1
            res, svd = self.measure(p[0], p[1], ch)
            if res == "near":
                self.clear(p[0], p[1], ch)
                return True
            if res == "direction":
                return True
        return False

    def _perp_ray(self, ch):
        """补第二条视线：从已有视线的**测点**出发，沿「射线前进 a + 侧偏 w」取点。

        候选按「离当前位置由近到远」排序先试近的；单线时不使用可行域中心族
        （那时可行域是一整个大楔形、中心无信息，实测会白跑约 200 s）。
        """
        rays = self.rays.get(ch, [])
        if not rays:
            return False
        a_cands, c_cands = [], []
        for (x, y, th) in list(rays)[-2:]:
            S = (x, y)
            u = uvec(th)
            n = perp(th)
            for (a, w) in ((500.0, 250.0), (500.0, -250.0), (300.0, 250.0),
                           (300.0, -250.0), (150.0, 150.0), (150.0, -150.0)):
                p = clamp((S[0] + u[0] * a + n[0] * w, S[1] + u[1] * a + n[1] * w))
                a_cands.append((dist(p, self.pos), p))
        reg = self._region(ch)
        if reg is not None and len(rays) >= 2 and self._has_good_pair(ch):
            cu = uvec(rays[-1][2])
            cn = perp(rays[-1][2])
            cxx, cyy = reg["c"]
            for (a, w) in ((120.0, 0.0), (-80.0, 0.0), (100.0, 150.0), (100.0, -150.0),
                           (200.0, 150.0), (200.0, -150.0), (250.0, 0.0),
                           (0.0, 150.0), (0.0, -150.0)):
                p = clamp((cxx + cu[0] * a + cn[0] * w, cyy + cu[1] * a + cn[1] * w))
                c_cands.append((dist(p, self.pos), p))
        a_cands.sort(key=lambda t: t[0])
        c_cands.sort(key=lambda t: t[0])
        cands = c_cands[:4] + a_cands
        tried, seen = 0, []
        for (_d, p) in cands:
            if tried >= 6:                     # 单次补线上限，避免为顽固源到处跑
                break
            if any(dist(p, q) < 60.0 for q in seen):
                continue
            seen.append(p)
            tried += 1
            if ch in self.cleared or not self._budget_ok():
                return ch in self.cleared
            res, svd2 = self.measure(p[0], p[1], ch)
            if res == "near":
                self.clear(p[0], p[1], ch)
                return ch in self.cleared
            if res == "direction" and svd2 is not None:
                if self._has_good_pair(ch):
                    return True
        return False

    def _region_clear(self, ch):
        """可行域格点清除：可行域多边形 poly 由各测向的 +/-1° 楔形与 <=1500 m 距离约束
        精确裁出 => 真值必在 poly 内。用间距 REG_S 的方格点阵（覆盖半径 sqrt2/2*间距 <20 m）
        铺满 poly 并逐点 /clear：/clear 只看距离、与朝向无关，失败仅 3 s，故铺满必命中。
        专治「源就在几米外、但该点测出 no_signal」而被迫长距离绕行的情形。
        """
        reg = self._region(ch)
        if reg is None or reg["R"] > V["REG_R"]:
            return False
        poly, c = reg["poly"], reg["c"]
        if self.clear(c[0], c[1], ch) == "success":
            return True
        xs = [p[0] for p in poly]
        ys = [p[1] for p in poly]
        x0, y0, x1, y1 = min(xs), min(ys), max(xs), max(ys)
        nx = int((x1 - x0) / V["REG_S"]) + 2
        ny = int((y1 - y0) / V["REG_S"]) + 2
        cand = []
        for j in range(ny):
            yy = y0 + j * V["REG_S"]
            for i in range(nx):
                p = (x0 + i * V["REG_S"], yy)
                if _poly_min_dist(p, poly) <= V["REG_E"]:
                    cand.append(p)
        cand.sort(key=lambda p: dist(p, self.pos))   # 近的优先，省移动
        for p in cand[:V["REG_PTS"]]:
            if ch in self.cleared:
                return True
            if not self._budget_ok():
                return False
            if self.clear(p[0], p[1], ch) == "success":
                return True
        return False

    def _ray_bracket_clear(self, ch, max_d=1500.0, step=100.0):
        """沿最新视线向源推进，用「有信号->无信号」的翻转点二分定出源距，再清除。

        全向源在视线上的信号集合是一段（进入有效半径到越过源），翻转点即源距；
        对定向源同样成立的补强（此处保留以便 Q3/Q4 工程一致）。
        """
        rays = self.rays.get(ch, [])
        if not rays:
            return False
        x0, y0, svd = rays[-1]
        ux, uy = uvec(svd)
        first = last_sig = None
        hi = None
        ds, d = [], 0.0
        while d < 400.0 - 1e-9:              # 近段 40 m 细步（源可能就在锚点附近）
            d += 40.0
            ds.append(d)
        while d < max_d - 1e-9:              # 远段 100 m 粗步
            d = min(max_d, d + step)
            ds.append(d)
        for d in ds:
            if not self._budget_ok():
                return ch in self.cleared
            p = clamp((x0 + d * ux, y0 + d * uy))
            res, _sv = self.measure(p[0], p[1], ch)
            if res == "near":
                self.clear(p[0], p[1], ch)
                return ch in self.cleared
            if res == "direction":
                if first is None:
                    first = d
                last_sig = d
            elif first is not None:
                hi = d
                break
        if first is None:
            return False
        if hi is None:
            hi = max_d
        lo = last_sig if last_sig is not None else first
        for _ in range(7):
            mid = 0.5 * (lo + hi)
            if not self._budget_ok():
                break
            p = clamp((x0 + mid * ux, y0 + mid * uy))
            res, _sv = self.measure(p[0], p[1], ch)
            if res == "near":
                self.clear(p[0], p[1], ch)
                return ch in self.cleared
            if res == "direction":
                lo = mid
            else:
                hi = mid
        for d2 in (lo, 0.5 * (lo + hi), hi):
            p = clamp((x0 + d2 * ux, y0 + d2 * uy))
            if self.clear(p[0], p[1], ch) == "success":
                return True
        return False

    def _home_clear(self, ch):
        ls = self.rays.get(ch, [])
        if not ls:
            return False
        p = self.pos
        last_good = p
        for step in (200.0, 140.0, 90.0, 60.0, 40.0, 25.0, 15.0):
            if ch in self.cleared or not self._budget_ok():
                return ch in self.cleared
            res, svd = self.measure(p[0], p[1], ch)
            if res == "near":
                self.clear(p[0], p[1], ch)
                return ch in self.cleared
            if res != "direction" or svd is None:
                mid = ((p[0] + last_good[0]) / 2.0, (p[1] + last_good[1]) / 2.0)
                for q in (last_good, mid):
                    if self.clear(q[0], q[1], ch) == "success":
                        return True
                return ch in self.cleared
            last_good = p
            p = (p[0] + step * math.cos(math.radians(svd)),
                 p[1] + step * math.sin(math.radians(svd)))
        return self.clear(p[0], p[1], ch) == "success"

    def _grid_clear(self, ch, cx, cy, rmax=GRID_RMAX, spacing=GRID_SPACING):
        if self.clear(cx, cy, ch) == "success":
            return True
        n = int(rmax / spacing)
        for ring in range(1, n + 1):
            r = ring * spacing
            pts = []
            for i in range(-ring, ring + 1):
                pts.append((cx + i * spacing, cy + r))
                pts.append((cx + i * spacing, cy - r))
            for j in range(-ring + 1, ring):
                pts.append((cx + r, cy + j * spacing))
                pts.append((cx - r, cy + j * spacing))
            for (x, y) in pts:
                if ch in self.cleared:
                    return True
                if not self._budget_ok():
                    return False
                if self.clear(x, y, ch) == "success":
                    return True
        return False

    # ==================== 联合路径规划 ====================
    @staticmethod
    def _pos_of(node, est):
        if node[0] == "pt":
            return COVER_POINTS[node[1]]
        return est.get(node[1]) or (0.0, 0.0)

    def _run_route(self):
        """固定覆盖骨架 + 源的最小代价插入 + 2-opt/Or-opt，按路线执行。"""
        start = (0.0, 0.0)
        cover_order = plan_order(start, list(range(len(COVER_POINTS))),
                                 lambda i: COVER_POINTS[i])
        route = [("pt", i) for i in cover_order]
        self.note("覆盖点开链顺序：%s" % [round(COVER_POINTS[i][0]) for i in cover_order[:4]])
        est = {}
        in_route = set()
        guard = 0
        while route and self._budget_ok() and guard < 500:
            guard += 1
            for ch in range(1, MAX_CH + 1):
                if (ch in self.cleared or ch in self.no_source
                        or ch in self._clear_failed):
                    continue
                reg = self._region(ch)
                if reg is None or reg["R"] > TUNE["INS_R"]:
                    continue
                est[ch] = reg["c"]
                if ch not in in_route:
                    in_route.add(ch)
                    self._insert(route, ("src", ch), est)
            pos_of = lambda n: self._pos_of(n, est)
            route = [route[i] for i in plan_order(self.pos, route, pos_of)]
            node = route.pop(0)
            if node[0] == "pt":
                self._scan_point(COVER_POINTS[node[1]])
            else:
                ch = node[1]
                in_route.discard(ch)
                if ch in self.cleared:
                    continue
                rr = self._region(ch)
                if rr is not None:
                    est[ch] = rr["c"]     # 用本频道的可行域中心更新估计（勿用上一轮残留值）
                self._resolve(ch)
                if ch not in self.cleared:
                    self.note("频道%d 暂未清除，稍后重试" % ch)
        self._visited_all = all(n[0] != "pt" for n in route)

    def _insert(self, route, node, est):
        """最小插入代价（含起点）把 node 插入 route。"""
        pos_of = lambda n: self._pos_of(n, est)
        if not route:
            route.append(node)
            return
        base = dist((0.0, 0.0), pos_of(route[0]))
        for a, b in zip(route, route[1:]):
            base += dist(pos_of(a), pos_of(b))
        best_i, best_c = len(route), None
        for i in range(len(route) + 1):
            if i == 0:
                c = (base - dist((0.0, 0.0), pos_of(route[0]))
                     + dist((0.0, 0.0), pos_of(node)) + dist(pos_of(node), pos_of(route[0])))
            elif i == len(route):
                c = base + dist(pos_of(route[-1]), pos_of(node))
            else:
                c = (base - dist(pos_of(route[i - 1]), pos_of(route[i]))
                     + dist(pos_of(route[i - 1]), pos_of(node))
                     + dist(pos_of(node), pos_of(route[i])))
            if best_c is None or c < best_c:
                best_i, best_c = i, c
        route.insert(best_i, node)

    # ==================== 主流程 ====================
    def run(self):
        t0 = time.monotonic()
        self.note("覆盖构型：中心 + %d*ρ=%.0fm，覆盖半径 %.1f m（<=%.0f）"
                  % (COVER_N, COVER_R, coverage_radius(step=30.0), R_MIN))
        self.enter()
        try:
            self._run_route()
            if self._visited_all:
                for ch in range(1, MAX_CH + 1):
                    if (ch not in self.cleared and ch not in self.no_source
                            and not self.rays.get(ch)):
                        if (V["GUARD"] and len(self._cover_hits.get(ch, ()))
                                < len(COVER_POINTS)):
                            # 完备性护栏：「覆盖后判无源」的前提是该频道确实测遍了
                            # 全部覆盖点（覆盖率定理的适用条件）。没测遍就如实留作
                            # 未清除上报，绝不谎报"无干扰源"。
                            self.note("频道%d 未测遍全部覆盖点（%d/%d），暂不判无源"
                                      % (ch, len(self._cover_hits.get(ch, ())),
                                         len(COVER_POINTS)))
                            continue
                        self.no_source.add(ch)
                        self.note("频道%d 覆盖后始终无信号，判定无干扰源" % ch)
            # 收尾清除：默认按「就近」顺序（而非频道号顺序），避免在场地两端来回空跑
            left = [ch for ch in sorted(set(self.rays) - self.cleared)]
            if V["TAIL"] and len(left) > 1:
                est = {}
                for ch in left:
                    reg = self._region(ch)
                    est[ch] = reg["c"] if reg is not None else (0.0, 0.0)
                order = plan_order(self.pos, list(left), lambda ch: est[ch])
                left = [left[i] for i in order]
            for ch in left:
                if not self._budget_ok():
                    break
                if ch in self.cleared:
                    continue
                self._resolve(ch)
            stuck = [ch for ch in self.rays if ch not in self.cleared]
            if stuck:
                self.note("以下频道未清除: %s" % stuck)
        finally:
            try:
                reason = self.exit_()
            except Exception:
                reason = "exit_error"
            dt = time.monotonic() - t0
            n = len(self.cleared)
            self.note("结束(%s)：清除 %d 个，虚拟时间 %.1f s，程序运行 %.1f s，平均 %.1f s"
                      % (reason, n, self.vt, dt, (self.vt / n) if n else 0.0))
        return {"cleared": sorted(self.cleared), "virtual_time": self.vt,
                "program_time": time.monotonic() - t0}
